### File 1: `src/hooks/useHasHydrated.ts`

Per D-09. This hook returns false during SSR and on the first client render, then
true after mount. Every component that renders cart-dependent data MUST gate on this.

```typescript
import { useEffect, useState } from "react";

export function useHasHydrated() {
  const [hasHydrated, setHasHydrated] = useState(false);
  useEffect(() => {
    setHasHydrated(true);
  }, []);
  return hasHydrated;
}
```

### File 2: `src/store/cart.ts`

Per D-07 (cart items only, no UI state), D-08 (localStorage key: 'rodo-cart'), D-10
(CartItem shape from types/index.ts). Import CartItem from `@/types`.

```typescript
import { create } from "zustand";
import { persist } from "zustand/middleware";
import type { CartItem } from "@/types";

interface CartStore {
  items: CartItem[];
  addItem: (item: CartItem) => void;
  removeItem: (productId: string, variantLabel: string | null) => void;
  updateQuantity: (productId: string, variantLabel: string | null, qty: number) => void;
  clearCart: () => void;
}

export const useCartStore = create<CartStore>()(
  persist(
    (set) => ({
      items: [],

      addItem: (item) =>
        set((state) => {
          const existing = state.items.find(
            (i) =>
              i.productId === item.productId &&
              i.variantLabel === item.variantLabel &&
              i.prepOption === item.prepOption
          );
          if (existing) {
            return {
              items: state.items.map((i) =>
                i.productId === item.productId &&
                i.variantLabel === item.variantLabel &&
                i.prepOption === item.prepOption
                  ? {
                      ...i,
                      quantity: i.quantity + item.quantity,
                      subtotalNgn: (i.quantity + item.quantity) * i.unitPriceNgn,
                    }
                  : i
              ),
            };
          }
          return { items: [...state.items, item] };
        }),

      removeItem: (productId, variantLabel) =>
        set((state) => ({
          items: state.items.filter(
            (i) =>
              !(i.productId === productId && i.variantLabel === variantLabel)
          ),
        })),

      updateQuantity: (productId, variantLabel, qty) =>
        set((state) => ({
          items: qty <= 0
            ? state.items.filter(
                (i) =>
                  !(i.productId === productId && i.variantLabel === variantLabel)
              )
            : state.items.map((i) =>
                i.productId === productId && i.variantLabel === variantLabel
                  ? { ...i, quantity: qty, subtotalNgn: qty * i.unitPriceNgn }
                  : i
              ),
        })),

      clearCart: () => set({ items: [] }),
    }),
    { name: "rodo-cart" }
  )
);
```

Notes on the store:
- `addItem` matches by productId + variantLabel + prepOption (a product with different
  prep options is a different line item).
- `updateQuantity` with qty <= 0 removes the item (defensive).
- No `isCartOpen` state — per D-07, UI state lives in React local state at the component
  level.

### File 1: `src/components/layout/Navbar.tsx`

Per D-13, a real shell component — not a comment placeholder. Uses `'use client'` because
it reads from the Zustand store via useCartStore and useHasHydrated.

```tsx
"use client";

import Link from "next/link";
import { ShoppingCart } from "lucide-react";
import { useCartStore } from "@/store/cart";
import { useHasHydrated } from "@/hooks/useHasHydrated";

export function Navbar() {
  const hasHydrated = useHasHydrated();
  const itemCount = useCartStore((s) => s.items.length);

  return (
    <header className="sticky top-0 z-50 border-b bg-white">
      <nav className="mx-auto flex h-16 max-w-7xl items-center justify-between px-4">
        <Link href="/" className="text-xl font-bold">
          Rodo &amp; Co
        </Link>

        <div className="flex items-center gap-6">
          <Link href="/shop" className="text-sm font-medium hover:underline">
            Shop
          </Link>

          <Link href="/checkout" className="relative" aria-label="Cart">
            <ShoppingCart className="h-5 w-5" />
            {hasHydrated && itemCount > 0 && (
              <span className="absolute -right-2 -top-2 flex h-4 w-4 items-center justify-center rounded-full bg-black text-[10px] font-bold text-white">
                {itemCount}
              </span>
            )}
          </Link>
        </div>
      </nav>
    </header>
  );
}
```

Notes:
- Cart icon links to /checkout for now (Phase 3 wires the cart sidebar).
- Badge only renders when `hasHydrated && itemCount > 0` — prevents hydration mismatch.
- No cart-open logic — per D-07, that is Phase 3 concern.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client localStorage -> Zustand store | Cart data read from untrusted storage on hydration |
| server render -> client hydrate | Mismatch between server HTML and client DOM causes React errors |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Tampering | localStorage 'rodo-cart' data | accept | Cart data is validated at checkout (Phase 5) against real DB prices; tampered localStorage only affects client display, not actual order totals |
| T-03-02 | Information Disclosure | Cart contents visible in localStorage | accept | No PII in cart — only productId, name, variant, quantity, price; low-value data |
| T-03-03 | Denial of Service | Hydration mismatch crashes React tree | mitigate | useHasHydrated hook gates all cart-dependent renders; badge shows nothing until hydration completes; verify step checks Navbar imports useHasHydrated |
| T-03-04 | Denial of Service | Missing @drawer/default.tsx causes 404 on hard refresh | mitigate | default.tsx created in same task as @drawer folder; verify step checks file exists and contains "return null"; npm run build validates slot wiring |
</threat_model>
