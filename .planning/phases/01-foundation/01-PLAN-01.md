### File 1: `src/types/env.d.ts`

Augment `NodeJS.ProcessEnv` with all 11 env vars. Any access to `process.env.FOO` for
an undeclared `FOO` will be a TypeScript error.

```typescript
// src/types/env.d.ts
// All environment variables required by Rodo & Co. Keep in sync with .env.local.example.
// NEXT_PUBLIC_* are safe in client bundles. Everything else is server-only.

declare namespace NodeJS {
  interface ProcessEnv {
    // ── Public (browser-safe) ──
    NEXT_PUBLIC_SUPABASE_URL: string;
    NEXT_PUBLIC_SUPABASE_ANON_KEY: string;
    NEXT_PUBLIC_PAYSTACK_PUBLIC_KEY: string;
    NEXT_PUBLIC_APP_URL: string;

    // ── Server-only (NEVER prefix with NEXT_PUBLIC_) ──
    SUPABASE_SERVICE_ROLE_KEY: string;
    PAYSTACK_SECRET_KEY: string;
    PAYSTACK_WEBHOOK_SECRET: string;
    RESEND_API_KEY: string;
    RESEND_FROM_EMAIL: string;
    ADMIN_NOTIFICATION_EMAIL: string;
    CRON_SECRET: string;

    // ── Standard Node ──
    NODE_ENV: "development" | "test" | "production";
  }
}

export {};
```

### File 2: `src/types/index.ts`

locked in D-10. All prices are `number` (integer kobo).

```typescript
// src/types/index.ts
// Shared domain types. DB column types mirrored from supabase/migrations/*.sql.
// All price fields are integer kobo (1 NGN = 100 kobo) — D-10.

// ── Products ──
export type ProductType = "fresh_produce" | "cooking_kit";

export interface Product {
  id: string;
  name: string;
  description: string | null;
  type: ProductType;
  image_url: string | null;
  is_active: boolean;
  created_at: string;
  updated_at: string;
}

export interface ProductVariant {
  id: string;
  product_id: string;
  label: string;
  price_ngn: number; // kobo
  is_active: boolean;
  created_at: string;
}

export interface PrepOption {
  id: string;
  product_id: string;
  label: string;
  is_active: boolean;
  created_at: string;
}

// ── Orders ──
export type OrderStatus = "pending" | "paid" | "processing" | "delivered";

export interface Order {
  id: string;
  paystack_reference: string;
  customer_name: string;
  customer_email: string;
  customer_phone: string;
  delivery_address: string;
  allergy_notes: string | null;
  total_ngn: number; // kobo
  status: OrderStatus;
  delivery_date: string | null;
  created_at: string;
  updated_at: string;
}

export interface OrderItem {
  id: string;
  order_id: string;
  product_id: string;
  variant_id: string | null;
  prep_option_id: string | null;
  product_name: string;
  variant_label: string | null;
  prep_option: string | null;
  quantity: number;
  unit_price_ngn: number; // kobo
  subtotal_ngn: number; // kobo
  created_at: string;
}

// ── Ordering config (single-row config table) ──
export interface OrderingConfig {
  id: 1;
  is_ordering_open: boolean;
  delivery_date: string | null;
  updated_at: string;
}

// ── Cart (client-side only; NEVER persisted in DB) ── D-10
export interface CartItem {
  productId: string;
  productName: string;
  variantLabel: string | null;
  prepOption: string | null;
  quantity: number;
  unitPriceNgn: number; // kobo
  subtotalNgn: number;  // kobo
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| developer laptop → git history | Real secrets committed accidentally |
| build pipeline → client bundle | Server-only env vars accidentally prefixed `NEXT_PUBLIC_` and shipped to browser |
| TypeScript compile → runtime | Undeclared env var access silently returns `undefined` |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Information Disclosure | `.env.local` contents | mitigate | `.gitignore` verified to include `.env.local`; verify step greps `.gitignore` |
| T-01-02 | Elevation of Privilege | `SUPABASE_SERVICE_ROLE_KEY` exposure via `NEXT_PUBLIC_` prefix | mitigate | `types/env.d.ts` declares the key without `NEXT_PUBLIC_`; verify step greps repo for any occurrence of `NEXT_PUBLIC_SUPABASE_SERVICE_ROLE_KEY` and fails if found |
| T-01-03 | Information Disclosure | Real values shipped in `.env.local.example` | mitigate | `.env.local.example` contains placeholder values only; code review before commit |
| T-01-04 | Tampering | Undeclared env var returns `undefined` at runtime | mitigate | All 11 env vars declared in `types/env.d.ts`; `npx tsc --noEmit` catches typos |
| T-01-05 | Spoofing | Wrong package installed (`framer-motion` vs `motion`) | mitigate | Task 1 verify step fails if `framer-motion` is present |
</threat_model>
