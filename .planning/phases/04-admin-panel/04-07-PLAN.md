       export function currentWeekOf(reference: Date = new Date()): string {
         const d = new Date(
           Date.UTC(
             reference.getUTCFullYear(),
             reference.getUTCMonth(),
             reference.getUTCDate(),
           ),
         );
         const dow = d.getUTCDay(); // 0 = Sunday
         d.setUTCDate(d.getUTCDate() - dow);
         const yyyy = d.getUTCFullYear();
         const mm = String(d.getUTCMonth() + 1).padStart(2, "0");
         const dd = String(d.getUTCDate()).padStart(2, "0");
         return `${yyyy}-${mm}-${dd}`;
       }
       ```

       const ngnFormatter = new Intl.NumberFormat("en-NG", {
         style: "currency",
         currency: "NGN",
         maximumFractionDigits: 0,
       });
       export function formatNgn(amount: number | string | null | undefined): string {
         if (amount === null || amount === undefined) return ngnFormatter.format(0);
         const n = typeof amount === "string" ? Number(amount) : amount;
         return ngnFormatter.format(Number.isFinite(n) ? n : 0);
       }
       ```
