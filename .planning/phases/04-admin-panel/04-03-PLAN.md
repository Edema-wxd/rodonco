       import "server-only";
       import { db } from "@/lib/db";
       import { orders, order_items } from "../../../drizzle/schema";
       import { desc, eq } from "drizzle-orm";

       export type AdminOrderItem = {
         id: string;
         product_name: string;
         variant_label: string | null;
         prep_option: string | null;
         quantity: number;
         unit_price_ngn: number;
         subtotal_ngn: number;
       };

       export type AdminOrder = {
         id: string;
         reference: string;
         customer_name: string;
         customer_email: string;
         customer_phone: string;
         delivery_address: string;
         allergy_notes: string | null;
         status: string;
         total_ngn: number;
         week_of: string;
         created_at: string;
         items: AdminOrderItem[];
       };

       export async function getAdminOrders(): Promise<AdminOrder[]> {
         // Two queries — neon-http doesn't do interactive transactions, but read consistency
         // here is best-effort and acceptable for an admin dashboard.
         const rows = await db.select().from(orders).orderBy(desc(orders.created_at));
         if (rows.length === 0) return [];
         const items = await db.select().from(order_items);
         const byOrderId = new Map<string, AdminOrderItem[]>();
         for (const it of items) {
           const list = byOrderId.get(it.order_id) ?? [];
           list.push({
             id: it.id,
             product_name: it.product_name,
             variant_label: it.variant_label,
             prep_option: it.prep_option,
             quantity: it.quantity,
             unit_price_ngn: it.unit_price_ngn,
             subtotal_ngn: it.subtotal_ngn,
           });
           byOrderId.set(it.order_id, list);
         }
         return rows.map((o) => ({
           id: o.id,
           reference: o.reference,
           customer_name: o.customer_name,
           customer_email: o.customer_email,
           customer_phone: o.customer_phone,
           delivery_address: o.delivery_address,
           allergy_notes: o.allergy_notes,
           status: o.status,
           total_ngn: o.total_ngn,
           week_of: typeof o.week_of === "string" ? o.week_of : new Date(o.week_of).toISOString().slice(0, 10),
           created_at: o.created_at instanceof Date ? o.created_at.toISOString() : String(o.created_at),
           items: byOrderId.get(o.id) ?? [],
         }));
       }

       import { NextResponse } from "next/server";
       import { auth } from "@/auth";
       import { db } from "@/lib/db";
       import { orders } from "../../../../../../drizzle/schema";
       import { eq } from "drizzle-orm";
       import { orderStatusPatchSchema } from "@/lib/admin/schemas";

       export async function PATCH(
         req: Request,
         { params }: { params: Promise<{ id: string }> },
       ) {
         const session = await auth();
         if (!session?.user) {
           return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
         }
         let body: unknown;
         try {
           body = await req.json();
         } catch {
           return NextResponse.json({ error: "Invalid JSON" }, { status: 400 });
         }
         const parsed = orderStatusPatchSchema.safeParse(body);
         if (!parsed.success) {
           return NextResponse.json(
             { error: "Invalid payload", issues: parsed.error.flatten() },
             { status: 400 },
           );
         }
         const { id } = await params;
         await db.update(orders).set({ status: parsed.data.status }).where(eq(orders.id, id));
         return NextResponse.json({ ok: true });
       }
       ```
       Note: Next.js 15 makes `params` a Promise; await it. Drizzle update is fine without transaction since we mutate one row.

       "use client";
       import { useState, useTransition } from "react";
       import { useRouter } from "next/navigation";
       import { toast } from "sonner";
       import {
         Select,
         SelectTrigger,
         SelectValue,
         SelectContent,
         SelectItem,
       } from "@/components/ui/select";

       const STATUS_OPTIONS = ["paid", "processing", "delivered"] as const;
       type Status = (typeof STATUS_OPTIONS)[number];

       export function OrderStatusSelect({
         orderId,
         initial,
       }: {
         orderId: string;
         initial: string;
       }) {
         const [value, setValue] = useState<string>(initial);
         const [, startTransition] = useTransition();
         const router = useRouter();

         async function handleChange(next: string) {
           const previous = value;
           setValue(next); // optimistic
           const res = await fetch(`/api/admin/orders/${orderId}`, {
             method: "PATCH",
             headers: { "Content-Type": "application/json" },
             body: JSON.stringify({ status: next }),
           });
           if (!res.ok) {
             setValue(previous);
             toast.error("Failed to update status. Please try again.");
             return;
           }
           startTransition(() => router.refresh());
         }

         return (
           <Select value={value} onValueChange={handleChange}>
             <SelectTrigger
               aria-label="Update order status"
               className="w-36"
               onClick={(e) => e.stopPropagation()}
             >
               <SelectValue />
             </SelectTrigger>
             <SelectContent>
               {STATUS_OPTIONS.map((s) => (
                 <SelectItem key={s} value={s}>
                   {s.charAt(0).toUpperCase() + s.slice(1)}
                 </SelectItem>
               ))}
             </SelectContent>
           </Select>
         );
       }
       ```
