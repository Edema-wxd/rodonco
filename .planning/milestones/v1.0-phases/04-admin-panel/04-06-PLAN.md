       ```typescript
       import { NextResponse } from "next/server";
       import { auth } from "@/auth";
       import { db } from "@/lib/db";
       import { ordering_config } from "../../../../../drizzle/schema";
       import { eq } from "drizzle-orm";
       import { orderingConfigPatchSchema } from "@/lib/admin/schemas";

       export async function PATCH(req: Request) {
         const session = await auth();
         if (!session?.user) {
           return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
         }
         let body: unknown;
         try {
           body = await req.json();
         } catch {
           return NextResponse.json({ error: "Invalid JSON" }, { status: 400 });
         }
         const parsed = orderingConfigPatchSchema.safeParse(body);
         if (!parsed.success) {
           return NextResponse.json(
             { error: "Invalid payload", issues: parsed.error.flatten() },
             { status: 400 },
           );
         }

         // Pitfall 6: updated_at is NOT auto-bumped on UPDATE — set it explicitly.
         await db
           .update(ordering_config)
           .set({
             is_ordering_open: parsed.data.is_ordering_open,
             updated_at: new Date(),
           })
           .where(eq(ordering_config.id, 1));

         return NextResponse.json({ ok: true });
       }
       ```

       ```typescript
       "use client";
       import { useEffect, useState } from "react";
       import { useRouter } from "next/navigation";
       import { toast } from "sonner";
       import { Card, CardContent } from "@/components/ui/card";
       import { Switch } from "@/components/ui/switch";
       import { Label } from "@/components/ui/label";
       import { Button } from "@/components/ui/button";

       export function OrderingToggle({ initialIsOpen }: { initialIsOpen: boolean }) {
         const router = useRouter();
         const [isOpen, setIsOpen] = useState(initialIsOpen);
         const [pendingClose, setPendingClose] = useState(false);
         const [savedVisible, setSavedVisible] = useState(false);
         const [submitting, setSubmitting] = useState(false);

         useEffect(() => {
           if (!savedVisible) return;
           const t = setTimeout(() => setSavedVisible(false), 2000);
           return () => clearTimeout(t);
         }, [savedVisible]);

         async function patchConfig(nextValue: boolean) {
           setSubmitting(true);
           const res = await fetch("/api/admin/config", {
             method: "PATCH",
             headers: { "Content-Type": "application/json" },
             body: JSON.stringify({ is_ordering_open: nextValue }),
           });
           setSubmitting(false);
           if (!res.ok) {
             toast.error("Failed to save settings.");
             // Revert local state
             setIsOpen(!nextValue);
             setPendingClose(false);
             return;
           }
           setIsOpen(nextValue);
           setPendingClose(false);
           setSavedVisible(true);
           router.refresh();
         }

         function handleSwitchChange(next: boolean) {
           if (!next) {
             // Closing — require confirmation; do NOT call API yet.
             setPendingClose(true);
             setIsOpen(false); // visual mirror; will revert on cancel
             return;
           }
           // Opening — call API immediately.
           setIsOpen(true);
           patchConfig(true);
         }

         function cancelClose() {
           setPendingClose(false);
           setIsOpen(true);
         }

         return (
           <Card>
             <CardContent className="space-y-4 p-6">
               <h2 className="text-xl font-semibold text-gray-900">Ordering Window</h2>
               <div
                 className="text-[28px] font-semibold leading-tight"
                 style={{ color: isOpen ? "#15803d" /* text-green-700 */ : "#b91c1c" /* text-red-700 */ }}
                 aria-live="polite"
               >
                 {isOpen ? "✓ Ordering is OPEN" : "✗ Ordering is CLOSED"}
               </div>
               <div className="flex items-center justify-between rounded-md border p-3">
                 <Label htmlFor="ordering-switch" className="text-sm font-medium">Enable ordering</Label>
                 <Switch
                   id="ordering-switch"
                   checked={isOpen}
                   onCheckedChange={handleSwitchChange}
                   disabled={submitting}
                   aria-label="Toggle ordering window"
                 />
               </div>

               {pendingClose ? (
                 <div className="space-y-2 rounded-md border border-destructive/30 bg-red-50 p-3">
                   <p className="text-sm text-destructive" id="close-confirm-text">
                     This will prevent new orders. Confirm?
                   </p>
                   <div className="flex gap-2">
                     <Button
                       type="button"
                       size="sm"
                       variant="destructive"
                       disabled={submitting}
                       onClick={() => patchConfig(false)}
                       aria-describedby="close-confirm-text"
                     >
                       Yes, Close
                     </Button>
                     <Button
                       type="button"
                       size="sm"
                       variant="ghost"
                       disabled={submitting}
                       onClick={cancelClose}
                     >
                       Cancel
                     </Button>
                   </div>
                 </div>
               ) : null}

               <div className="flex items-center justify-between pt-2">
                 <Button
                   type="button"
                   onClick={() => patchConfig(isOpen)}
                   disabled={submitting || pendingClose}
                 >
                   Save Settings
                 </Button>
                 <span
                   className={
                     "text-sm text-green-600 transition-opacity duration-1000 " +
                     (savedVisible ? "opacity-100" : "opacity-0")
                   }
                   role="status"
                   aria-hidden={!savedVisible}
                 >
                   Saved
                 </span>
               </div>
             </CardContent>
           </Card>
         );
       }
       ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → PATCH /api/admin/config | Untrusted boolean toggle; affects entire customer ordering ability |
| Browser → /admin/settings page | Authenticated session required |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-27 | Elevation of Privilege | PATCH /api/admin/config | mitigate | `auth()` at top → 401 if no session; middleware also enforces. |
| T-04-28 | Tampering | Config PATCH body | mitigate | `orderingConfigPatchSchema.safeParse` with `.strict()` rejects extra fields and non-boolean is_ordering_open. |
| T-04-29 | Tampering | Config row id manipulation | mitigate | Handler hardcodes `where(eq(ordering_config.id, 1))` — id is not user-controlled; cannot target a different row. |
| T-04-30 | Repudiation | Who closed ordering and when? | mitigate | `updated_at: new Date()` is bumped on every PATCH (Pitfall 6). Audit trail of WHO did it would require an audit_log table — out of scope for v1; accept as residual risk because there is exactly one admin role at MVP. |
| T-04-31 | Denial of Service | Accidental ordering closure | mitigate | Inline confirmation ("This will prevent new orders. Confirm?") blocks accidental clicks; opening is one-click (low-risk direction). |
</threat_model>
